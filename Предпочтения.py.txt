import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

plt.style.use('dark_background')
np.random.seed(142) 
banks = ['Сбербанк', 'ВТБ', 'Тинькофф', 'Альфа-Банк', 'Газпромбанк', 'Райффайзенбанк']
n_respondents = 10000
data = {
    'Возраст': np.random.randint(18, 80, n_respondents),
    'Пол': np.random.choice(['Мужской', 'Женский'], n_respondents),
    'Город': np.random.choice(['Москва', 'СПб', 'Регионы'], n_respondents, p=[0.3, 0.2, 0.5]),
    'Предпочтительный банк': np.random.choice(banks, n_respondents, p=[0.35, 0.20, 0.15, 0.12, 0.10, 0.08])
}

df = pd.DataFrame(data)
print("Первые 10 строк данных опроса:")
print(df.head(10))
print("\n" + "="*50 + "\n")
print("ОПИСАТЕЛЬНАЯ СТАТИСТИКА:")
print(f"Общее количество респондентов: {len(df)}")
print(f"Возраст: средний = {df['Возраст'].mean():.1f} лет, "
      f"медиана = {df['Возраст'].median():.0f} лет, "
      f"диапазон = {df['Возраст'].min()}-{df['Возраст'].max()} лет")
print("\nРаспределение по полу:")
print(df['Пол'].value_counts())
print("\nРаспределение по городам:")
print(df['Город'].value_counts())
print("\n" + "="*50 + "\n")
print("ПРЕДПОЧТЕНИЯ ПО БАНКАМ:")
bank_preferences = df['Предпочтительный банк'].value_counts()
bank_percentages = df['Предпочтительный банк'].value_counts(normalize=True) * 100
results = pd.DataFrame({
    'Количество голосов': bank_preferences,
    'Процент': bank_percentages.round(2)
})
print(results)
print(f"\nСамый популярный банк: {bank_preferences.index[0]} "
      f"({bank_preferences.iloc[0]} голосов, {bank_percentages.iloc[0]:.2f}%)")
print("\n" + "="*50 + "\n")
print("АНАЛИЗ ПО ГРУППАМ:")
print("\nПредпочтения по полу:")
gender_analysis = pd.crosstab(df['Предпочтительный банк'], df['Пол'], normalize='columns') * 100
print(gender_analysis.round(2))
df['Возрастная группа'] = pd.cut(df['Возраст'], bins=[18, 30, 45, 60, 80], labels=['18-30', '31-45', '46-60', '61+'])
age_analysis = pd.crosstab(df['Предпочтительный банк'], df['Возрастная группа'], normalize='columns') * 100
print("\nПредпочтения по возрастным группам (%):")
print(age_analysis.round(2))
city_analysis = pd.crosstab(df['Предпочтительный банк'], df['Город'], normalize='columns') * 100
print("\nПредпочтения по городам (%):")
print(city_analysis.round(2))

fig, axes = plt.subplots(2, 2, figsize=(15, 12))
fig.suptitle('Результаты социологического опроса: предпочтения банков', fontsize=16, fontweight='bold')
ax1 = axes[0, 0]
colors1 = plt.cm.Set3(np.linspace(0, 1, len(banks)))
wedges, texts, autotexts = ax1.pie(bank_preferences, labels=bank_preferences.index, autopct='%1.1f%%', colors=colors1)
ax1.set_title('Распределение предпочтений по банкам', fontsize=12, fontweight='bold')
ax2 = axes[0, 1]
bars = ax2.bar(bank_preferences.index, bank_preferences.values, color=colors1)
ax2.set_title('Количество голосов за каждый банк', fontsize=12, fontweight='bold')
ax2.set_ylabel('Количество респондентов')
for bar in bars:
    height = bar.get_height()
    ax2.text(bar.get_x() + bar.get_width()/2., height + 10, f'{int(height)}',
              ha='center', va='bottom', fontsize=9)
ax3 = axes[1, 0]
age_pivot = pd.crosstab(df['Возрастная группа'], df['Предпочтительный банк'])
age_pivot.plot(kind='bar', stacked=True, ax=ax3, color=colors1, edgecolor='black')
ax3.set_title('Предпочтения по возрастным группам', fontsize=12, fontweight='bold')
ax3.set_xlabel('Возрастная группа')
ax3.set_ylabel('Количество респондентов')
ax3.legend(title='Банк', bbox_to_anchor=(1.05, 1), loc='upper left')
ax4 = axes[1, 1]
city_pivot = pd.crosstab(df['Город'], df['Предпочтительный банк'])
city_pivot.plot(kind='bar', stacked=True, ax=ax4, color=colors1, edgecolor='black')
ax4.set_title('Предпочтения по городам', fontsize=12, fontweight='bold')
ax4.set_xlabel('Город')
ax4.set_ylabel('Количество респондентов')
ax4.legend(title='Банк', bbox_to_anchor=(1.05, 1), loc='upper left')
plt.tight_layout()
plt.show()
