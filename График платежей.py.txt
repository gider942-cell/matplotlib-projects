import numpy as np 
import pandas as pd 
import matplotlib.pyplot as plt 
from datetime import timedelta, date 

np.random.seed(42) 

start_date = date.today() + timedelta(days=1)
dates = pd.date_range(start=start_date, periods=30, freq='D') 

amounts = np.random.randint(100, 5000, size=len(dates))

is_weekend = dates.weekday >= 5 
amounts[is_weekend] = np.maximum(amounts[is_weekend] - 1000, 100) 

df = pd.DataFrame({
    'date': dates,
    'amount': amounts
})

categories = ['Аренда', 'Зарплата', 'Закупка', 'Прочие']
df['category'] = np.random.choice(categories, size=len(df))

df = df.sort_values('date').reset_index(drop=True)

daily_totals = df.groupby('date')['amount'].sum().reset_index()

plt.style.use('seaborn-v0_8-whitegrid')
fig, ax = plt.subplots(figsize=(12, 6))

ax.bar(daily_totals['date'], daily_totals['amount'], color='#4c72b0', alpha=0.8)

ax.set_title('Планируемые платежи на следующий месяц', fontsize=16, pad=20)
ax.set_xlabel('Дата', fontsize=12)
ax.set_ylabel('Сумма платежа, ?', fontsize=12)

ax.xaxis.set_major_formatter(plt.matplotlib.dates.DateFormatter('%d.%m'))
plt.xticks(rotation=45)

plt.tight_layout()
plt.show()

print("Сводная статистика платежей:")
print(daily_totals.describe())
print("\nОбщая сумма платежей за месяц:", daily_totals['amount'].sum(), "?")
